# Gramática de prueba para recuperación de errores (LR(1))
# Terminales, no terminales, inicial y producciones.

TERMINALES: ID NUMBER STRING = + ( ) ; , bold italic color upper lower
NO_TERMINALES: Program StmtList Stmt Expr ExprTail Term Factor FuncCall FunctionName ArgList Literal
INICIAL: Program
PRODUCCIONES:
Program -> StmtList
StmtList -> StmtList Stmt | Stmt
Stmt -> Expr ;
Stmt -> ID = Expr ;
Expr -> Term ExprTail
ExprTail -> + Term ExprTail | ε
Term -> Factor
Factor -> FuncCall | ID | Literal | ( Expr )
FuncCall -> FunctionName ( ArgList )
ArgList -> Expr , ArgList | Expr | ε
FunctionName -> bold | italic | color | upper | lower
Literal -> NUMBER | STRING

# Inválido (token inesperado): resultado = color(@Rojo) ;
# Inválido (sin punto y coma): resultado = color("Rojo") + upper(nombre)
